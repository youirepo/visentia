# Visentia v0.1 runtime dependencies.
# Manim is pinned exactly because the StaticLinter's Manim CE API model (added in slice #7)
# must stay in sync with the rendering engine. Bumping Manim requires regenerating that model.

# Voiceover stack (slice #3). manim-voiceover 0.3.7 does not ship its own edge-tts service —
# Visentia provides one in src/visentia/voiceover.py (EdgeService).

edge-tts==7.2.8

# LLM tier. `anthropic` is the provider the engine uses — Sonnet 5 for classification and
# parameter filling, Opus 5 for freeform codegen and narration (issue #27).
anthropic==1.5.0

# google-genai predates the #27 model decision and is no longer wired into the
# orchestrator. Kept until the Gemini provider and its tests are removed.
google-genai==2.4.0
python-dotenv==0.21.1

# Web UI (slice #5)
fastapi==0.115.12
uvicorn[standard]==0.34.2

# setuptools<80 is a workaround: manim-voiceover 0.3.7 imports `pkg_resources` at module
# load time, and setuptools 80 (released 2025) removed it. Drop this pin when
# manim-voiceover ships a fix.
setuptools<80
